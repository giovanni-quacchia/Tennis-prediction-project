\lstdefinestyle{BaseProgrammingStyle}{
  keywordstyle=\color[HTML]{d8127e}\bfseries,   % keyword: bf pink
  commentstyle=\color[HTML]{009900},            % comments: green
  backgroundcolor=\color[HTML]{f2f2eb},         % Background: beige   
  numberstyle=\color{gray},                     % line numbers: gray
  stringstyle=\color[HTML]{9400d1},             % strings: purple
  aboveskip=1em,                          
  belowskip=1em,
  upquote=true,
  numbers=left,  
  basicstyle=\ttfamily\small,
  columns=fullflexible,
  frame=none,                             
  showstringspaces=false, 
  breaklines=true, 
  breakatwhitespace=true, 
  lineskip=0pt,
  sensitive=false,
  literate={~} {$\sim$}{1}                      % tilde classico
}
